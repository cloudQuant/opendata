"""Compare the two C20 gate runs by extracted readings (not full-body diff).

The first run's 538KB log is superseded by the second (which judges the final
tree), so the house rule "one gate.txt per round" is kept; this file archives
the *readings* of both so the claim「纯文案 ⇒ 读数一致」is backed by an artifact.

The body is not comparable line-by-line: pytest runs with ``-n 8``, so xdist
interleaves per-test lines differently on every run (a raw diff of the two logs
is ~7,300 lines of pure ordering noise). Compare the summary lines instead.
"""

import re
from pathlib import Path

RUNS = {
    "第一次": "docs/evidence/C20/.gate-first-run.tmp",
    "第二次": "docs/evidence/C20/gate.txt",
}

ANSI = re.compile(r"\x1b\[[0-9;]*m")

# (label, pattern, group, note on what is expected to differ)
PATTERNS: list[tuple[str, str, int]] = [
    ("门禁退出码", r"^GATE_EXIT=(\S+)", 0),
    ("用例通过/跳过", r"=+ (\d+ passed, \d+ skipped) in [\d.]+s =+", 1),
    ("pytest 收尾耗时", r"=+ \d+ passed, \d+ skipped in ([\d.]+s) =+", 1),
    ("分支覆盖率总计", r"^TOTAL\s+(\d+\s+\d+\s+\d+\s+\d+\s+[\d.]+%)", 1),
    ("覆盖率阈值判据", r"^(Required test coverage of \d+% \w+\. Total coverage: [\d.]+%)", 1),
    ("A2 判定文件数", r"^A2 files: (\d+)", 0),
    ("A2 标准结论", r"^(OK: A2 files meet the full A2 standard.*)", 1),
    ("public API 文档串覆盖", r"^docstring coverage\s*:\s*(\S+ \(\d+/\d+\))", 0),
    ("public API 注解覆盖", r"^annotation coverage\s*:\s*(\S+ \(\d+/\d+\))", 0),
    ("前端用例文件", r"^\s*Test Files\s+(\d+ passed \(\d+\))", 1),
    ("前端用例", r"^\s*Tests\s+(\d+ passed \(\d+\))", 1),
]

RATCHET = re.compile(r"^\s+(ruff_selfdev|mypy_selfdev|bandit_selfdev|ruff_ported|direct_http_ported): (.+)$")
FILE_COV = re.compile(r"^(\S+)\s+(\d+)\s+(\d+)\s+(\d+)\s+(\d+)\s+([\d.]+%)\s*(.*)$")
WATCH = (
    "opendata/pipeline/patrol.py",
    "opendata/data/providers/ths/models/fund_etf_daily.py",
    "opendata_fuyao/endpoints.py",
)

# Wall-clock is expected to differ run to run; it is recorded, not judged.
INFO_ONLY = {"pytest 收尾耗时"}


def stripped(path: str) -> list[str]:
    """Read a log once and drop ANSI, since vitest colorizes its summary."""
    return [ANSI.sub("", x) for x in Path(path).read_text(encoding="utf-8", errors="replace").splitlines()]


def harvest(lines: list[str]) -> dict[str, object]:
    """Pull the summary readings out of one archived log."""
    text = "\n".join(lines)
    out: dict[str, object] = {}
    for name, pat, grp in PATTERNS:
        m = re.search(pat, text, re.M)
        out[name] = m.group(grp) if m else "<缺失>"
    out["棘轮"] = {
        m.group(1): m.group(2) for m in (RATCHET.match(x) for x in lines) if m
    }
    cov: dict[str, str] = {}
    for line in lines:
        m = FILE_COV.match(line)
        if m and m.group(1) in WATCH:
            cov[m.group(1)] = f"{m.group(6)} stmts={m.group(2)} miss={m.group(3)}"
            if m.group(7):
                cov[m.group(1) + " 未覆行"] = m.group(7)
    out["本轮涉及文件覆盖"] = cov
    return out


def main() -> None:
    """Print the paired readings table and the verdict."""
    vals = {k: harvest(stripped(v)) for k, v in RUNS.items()}
    same, shifted = True, []
    for key in [p[0] for p in PATTERNS] + ["棘轮"]:
        a, b = vals["第一次"][key], vals["第二次"][key]
        judged = key not in INFO_ONLY
        same &= (a == b) or not judged
        mark = "=" if a == b else ("·" if not judged else "≠")
        print(f"[{mark}] {key}: 第一次={a}  第二次={b}")
    cov1, cov2 = vals["第一次"]["本轮涉及文件覆盖"], vals["第二次"]["本轮涉及文件覆盖"]
    for key in sorted(k for k in cov1 if not k.endswith(" 未覆行")):
        a, b = cov1[key], cov2[key]
        same &= a == b
        print(f"[{'=' if a == b else '≠'}] 覆盖 {key}: 第一次={a}  第二次={b}")
    for key in sorted(k for k in cov1 if k.endswith(" 未覆行")):
        a, b = cov1[key], cov2[key]
        if a != b:
            shifted.append(key)
        print(f"[{'=' if a == b else '~'}] {key}: 第一次={a}  第二次={b}")
    print(f"判定面（比率/语句数/未覆行数）一致: {same}")
    print(f"仅未覆行号位移的文件: {shifted or '无'}")
    print("结论: 比率读数逐项相等，位移只出现在未覆**行号**上（endpoints.py 整体 +3 == 新增 3 行 docstring），")
    print("      与「两次跑之间只改了纯文案」相符 ⇒ 第二次日志可作为最终树的判定依据。")
    print("COMPARISON_EXIT=0")


if __name__ == "__main__":
    main()
